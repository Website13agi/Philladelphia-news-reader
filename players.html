async function loadPlayers() { const container = document.getElementById( "players-container" ); const updated = document.getElementById( "roster-updated" ); try { updated.textContent = "Loading MLB roster..."; const response = await fetch( "players.json?ts=" + Date.now(), { cache: "no-store" } ); if (!response.ok) { throw new Error( "players.json HTTP " + response.status ); } const data = await response.json(); if ( !data || !Array.isArray( data.players ) ) { throw new Error( "Invalid players.json" ); } if ( data.players.length === 0 ) { throw new Error( "players.json contains no players" ); } renderPlayers(data); } catch (error) { console.error( "Phillies roster error:", error ); updated.textContent = "Roster unavailable"; container.innerHTML = `
Unable to load the Phillies roster.

players.json could not be loaded.
Please run the Update Phillies Roster GitHub Action.
`; } } loadPlayers();
